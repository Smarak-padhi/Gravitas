-- study_db_bootstrap.sql: Immutable Local Storage
CREATE TABLE courses (
  id TEXT PRIMARY KEY,
  catalog_code TEXT NOT NULL UNIQUE,
  title TEXT NOT NULL,
  credits REAL NOT NULL DEFAULT 4.0
);

CREATE TABLE verbatim_directives (
  id TEXT PRIMARY KEY,
  course_id TEXT REFERENCES courses(id),
  lecture_timestamp INTEGER NOT NULL,
  audio_sha256 BLOB,
  verbatim_instruction TEXT NOT NULL,
  immutable_audit_lock INTEGER DEFAULT 1
);

CREATE TABLE student_working_drafts (
  id TEXT PRIMARY KEY,
  directive_id TEXT REFERENCES verbatim_directives(id),
  draft_markdown TEXT NOT NULL,
  updated_at INTEGER NOT NULL
);
